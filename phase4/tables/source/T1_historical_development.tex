\toprule
\textbf{Theory} & \textbf{Year} & \textbf{Primary Source} & \textbf{Key Mathematical Contribution} & \textbf{Applicability} & \textbf{Evidence} \\
\midrule
Classical coupled (CD) & 1956 & Biot \cite{biot1956} & Classical coupled thermoelasticity with Fourier heat conduction $q_i=-kT_{,i}$ & Isotropic solids, small deformations, infinite thermal speed & \cite{biot1956} \\
Lord--Shulman (LS) & 1967 & Lord \& Shulman \cite{lord1967} & One relaxation time $\tau_0$; modified heat equation $q_i+\tau_0\dot{q}_i=-kT_{,i}$; hyperbolic heat transfer & Finite thermal wave speed $c_T=\sqrt{k/(\rho c_v\tau_0)}$ & \cite{lord1967} \\
Green--Lindsay (GL) & 1972 & Green \& Lindsay \cite{green1972} & Two relaxation times $\tau_0,\tau_1$; temperature-rate dependent constitutive relations & Finite wave speed; no dissipation for centrally symmetric bodies & \cite{green1972} \\
G--N Type II & 1991 & Green \& Naghdi \cite{green1991} & Thermal displacement $\alpha_i=\int T_i\,dt$; entropy flux from free energy potential; no energy dissipation & Wave propagation without dissipation; second sound & \cite{green1991} \\
G--N Type III & 1993 & Green \& Naghdi \cite{green1993} & Dissipative type III theory; mixed type I--II character & Finite wave speed with dissipation & \cite{green1993} \\
Dual-phase-lag (DPL) & 1995 & Tzou \cite{tzou1995} & Two phase lags $\tau_q$ (flux) and $\tau_T$ (gradient); $q(t+\tau_q)=-k\nabla T(t+\tau_T)$ & Micro-scale heat transfer; $\tau_q\ge\tau_T$ for admissibility & \cite{tzou1995} \\
Triple-phase-lag (TPL) & 1997 & Tzou \cite{tzou1997} & Three phase lags including thermal displacement gradient & Extended lag models for ultrafast processes & \cite{tzou1997} \\
Memory-dependent derivative (MDD) & 2011 & Wang \& Li \cite{wang2011} & Integral-form memory operator $D_\omega f(t)=\frac{1}{\omega}\int_{t-\omega}^t k(t-\xi)f'(\xi)\,d\xi$ & Hereditary memory effects; kernel-function based & \cite{wang2011} \\
Nonlocal elasticity & 1972 & Eringen \cite{eringen1972} & Integral constitutive relations $\sigma_{ij}=\int_\Omega A(|\mathbf{x}-\mathbf{x}'|)C_{ijkl}\varepsilon_{kl}(\mathbf{x}')\,d\mathbf{x}'$ & Size-dependent wave dispersion; length scale $(e_0a)^2$ & \cite{eringen1972} \\
Moore--Gibson--Thompson (MGT) & 2019 & Quintanilla \cite{quintanilla2019} & Third-order heat equation $\tau\ddot{q}+\dot{q}=-(k/\rho c)\nabla(\dot{T}+\tau\ddot{T})$ & Thermal inertia; wave propagation with relaxation & \cite{quintanilla2019} \\
